cmake_minimum_required(VERSION 3.20)
project(FetchContentFixture NONE)
include(FetchContent)
FetchContent_Declare(locked_source URL "${LOCKED_URL}" URL_HASH "SHA256=${LOCKED_SHA256}" SOURCE_SUBDIR no-such-subdirectory)
FetchContent_MakeAvailable(locked_source)
file(READ "${locked_source_SOURCE_DIR}/Source/CMakeVersion.cmake" version)
if(NOT version MATCHES "set\\(CMake_VERSION_MAJOR 4\\)" OR NOT version MATCHES "set\\(CMake_VERSION_MINOR 4\\)")
  message(FATAL_ERROR "fetched CMake source has an unexpected version")
endif()
