cmake_minimum_required(VERSION 3.20)
project(GraphFixture LANGUAGES C CXX)
include(GNUInstallDirs)
include(CheckCSourceCompiles)
include(CheckCXXSourceCompiles)
check_c_source_compiles("int main(void) { return 0; }" C_FEATURE_WORKS)
check_cxx_source_compiles("int main() { return 0; }" CXX_FEATURE_WORKS)
if(NOT C_FEATURE_WORKS OR NOT CXX_FEATURE_WORKS)
  message(FATAL_ERROR "feature checks failed")
endif()
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(FixtureMath MODULE REQUIRED)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/try.c" "int main(void) { return 0; }\n")
try_compile(COMPILES "${CMAKE_CURRENT_BINARY_DIR}/try-build" "${CMAKE_CURRENT_BINARY_DIR}/try.c")
try_run(RUNS COMPILED "${CMAKE_CURRENT_BINARY_DIR}/run-build" "${CMAKE_CURRENT_BINARY_DIR}/try.c")
if(NOT COMPILES OR NOT COMPILED OR NOT RUNS EQUAL 0)
  message(FATAL_ERROR "compiler checks failed")
endif()
add_library(imported INTERFACE IMPORTED)
set_target_properties(imported PROPERTIES INTERFACE_COMPILE_DEFINITIONS IMPORTED_VALUE=5)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated.c"
  COMMAND "${CMAKE_COMMAND}" -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/generated.c -P "${CMAKE_CURRENT_SOURCE_DIR}/generate.cmake"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generate.cmake" VERBATIM)
add_library(base STATIC base.c "${CMAKE_CURRENT_BINARY_DIR}/generated.c")
target_include_directories(base PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
target_link_libraries(base PRIVATE "$<BUILD_INTERFACE:imported>" "$<BUILD_INTERFACE:FixtureMath::FixtureMath>")
add_library(wrapper STATIC wrapper.cxx)
target_link_libraries(wrapper PUBLIC base)
target_compile_features(wrapper PUBLIC cxx_std_11)
target_compile_definitions(wrapper PRIVATE "$<$<CONFIG:Release>:RELEASE_BUILD=1>")
add_executable(graph main.cxx)
target_link_libraries(graph PRIVATE wrapper)
install(TARGETS base wrapper EXPORT GraphTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(EXPORT GraphTargets NAMESPACE Graph:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/Graph")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/GraphConfig.cmake" "include(\"\${CMAKE_CURRENT_LIST_DIR}/GraphTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/GraphConfig.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/Graph")
